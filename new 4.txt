name: STM32 Firmware Build Automation

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    # ૧. ગિટહબ સર્વર પર સોર્સ કોડ અને બધા જ સબમોડ્યુલ્સ (જેમ કે FreeRTOS) ઓટોમેટિક લાવવા માટે
    - name: Checkout Code
      uses: actions/checkout@v4
      with:
        submodules: 'recursive' # આ એક જ લાઈન ક્લાઉડમાં Middlewares/FreeRTOS આખું ડાઉનલોડ કરી દેશે

    # ૨. STM32F103 (Cortex-M3) માટે ARM ક્રોસ-કમ્પાઇલર ઇન્સ્ટોલ કરવા માટે
    - name: Install ARM Toolchain
      uses: carlosperate/arm-none-eabi-gcc-action@v1
      with:
        release: '10.3-2021.10'

    # ૩. કોડ કમ્પાઇલ કરીને ઓટોમેટિક HEX ફાઇલ જનરેટ કરવી (સબમોડ્યુલ સપોર્ટ સાથે)
    - name: Build Firmware Without Debug Folder
      shell: bash
      run: |
        set -euxo pipefail

        # ઓબ્જેક્ટ ફાઇલો સેવ કરવા માટે નવું ટેમ્પરરી ફોલ્ડર બનાવવું
        mkdir -p build_out
        
        # STM32F103 (Cortex-M3) માટેના સ્પેસિફિક ફ્લેગ્સ
        MCU_FLAGS="-mcpu=cortex-m3 -mthumb"
        
        # ગ્લોબલ મેક્રોઝ (-D_XOPEN_SOURCE=600 ઉમેરવાથી sysmem.c ની caddr_t વાળી એરર ડાયરેક્ટ સોલ્વ થઈ જશે)
        DEFINES="-DSTM32F103xB -DUSE_HAL_DRIVER -D_XOPEN_SOURCE=600"
        
        # બધા જ ઇન્ક્લુડ ફોલ્ડર્સ ઓટોમેટિક શોધવા (બધા જ બિનજરૂરી ફોલ્ડર્સ પરફેક્ટ બેકસ્લેશ લિંકિંગ સાથે ઇગ્નૉર કર્યા છે)
        INC_DIRS=$(find . -type d \
          -not -path './.git*' \
          -not -path './build_out*' \
          -not -path './Middlewares/FreeRTOS/portable*' \
          -not -path './Middlewares/FreeRTOS/examples*' \
          -not -path './Middlewares/FreeRTOS/.github*')

        INCLUDES=""
        while IFS= read -r dir; do
          if [ -n "$dir" ]; then
            INCLUDES="$INCLUDES -I$dir"
          fi
        done <<< "$INC_DIRS"
        
        # ખાસ FreeRTOS ના પોર્ટેબલ હેડર્સનો પાથ મેન્યુઅલી ઉમેરવો
        INCLUDES="$INCLUDES -IMiddlewares/FreeRTOS/portable/GCC/ARM_CM3"
        
        CFLAGS="$MCU_FLAGS $DEFINES $INCLUDES -O0 -g3 -Wall -ffunction-sections -fdata-sections"
        
        # બધી જ વેલિડ .c ફાઈલો લિસ્ટ કરવી
        mapfile -t SOURCES < <(
          find . -name '*.c' \
            -not -path './.git/*' \
            -not -path './build_out/*' \
            -not -path './Middlewares/FreeRTOS/portable/*' \
            -not -path './Middlewares/FreeRTOS/examples/*' \
            -not -path './Middlewares/FreeRTOS/.github/*'
        )
        
        # STM32F103 માટેની સાચી પોર્ટ ફાઈલ એડ કરવી
        # STM32F103 માટેની સાચી પોર્ટ ફાઈલ અને મેમરી મેનેજમેન્ટ (Heap) ફાઈલ એડ કરવી
        SOURCES+=( "./Middlewares/FreeRTOS/portable/GCC/ARM_CM3/port.c" )
        SOURCES+=( "./Middlewares/FreeRTOS/portable/MemMang/heap_4.c" )

        echo "કમ્પાઈલેશન શરૂ થઈ રહ્યું છે..."
        # દરેક .c ફાઇલ માટે યુનિક ઓબ્જેક્ટ .o ફાઇલો બનાવવી (ફોલ્ડર સ્ટ્રક્ચર સાથે જેથી ઓવરરાઈટ ન થાય)
        for file in "${SOURCES[@]}"; do
          # પાથના સ્લેશ (/) ને અંડરસ્કોર (_) માં બદલીને યુનિક નામ આપવું (જેમ કે Core_Src_main.o)
          safe_obj_name=$(echo "$file" | sed 's/[\/.]/_/g')
          obj="build_out/${safe_obj_name}.o"
          
          echo "Compiling: $file"
          arm-none-eabi-gcc $CFLAGS -c "$file" -o "$obj"
        done
        
        echo "લિંકિંગ પ્રોસેસ શરૂ થઈ રહી છે..."
        LINKER_SCRIPT=$(find . -name '*.ld' -not -path './.git/*' | head -n 1)
        if [ -z "$LINKER_SCRIPT" ]; then
          echo "ભૂલ: પ્રોજેક્ટમાં લિંકર સ્ક્રિપ્ટ (.ld ફાઇલ) મળી નથી!"
          exit 1
        fi
        
        echo "Using Linker Script: $LINKER_SCRIPT"
        
        # બધી ઓબ્જેક્ટ ફાઇલોને લિંક કરીને ફાઇનલ .elf ફાઇલમાં રૂપાંતર કરવું
        arm-none-eabi-gcc $MCU_FLAGS -T"$LINKER_SCRIPT" build_out/*.o -o build_out/firmware.elf -Wl,--gc-sections --specs=nano.specs -lm
        
        echo "ફાઇનલ .hex અને .bin ફાઇલો જનરેટ થઈ રહી છે..."
        arm-none-eabi-objcopy -O ihex build_out/firmware.elf build_out/firmware.hex
        arm-none-eabi-objcopy -O binary build_out/firmware.elf build_out/firmware.bin
        
        # ૪. જનરેટ થયેલી HEX અને BIN ફાઇલોને ગિટહબ આર્ટિફેક્ટ્સ પર અપલોડ કરવી
        echo "બિલ્ડ સફળ રહ્યું! ફાઇલો તૈયાર છે."

    # ૪. જનરેટ થયેલી HEX અને BIN ફાઇલોને ગિટહબ આર્ટિફેક્ટ્સ પર અપલોડ કરવી
    - name: Upload Output Firmware
      uses: actions/upload-artifact@v4
      with:
        name: STM32F103-Firmware
        path: |
          build_out/firmware.hex
          build_out/firmware.bin
